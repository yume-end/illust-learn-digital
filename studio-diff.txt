 function drawStroke(ctx: CanvasRenderingContext2D, stroke: Stroke, scaleX = 1, scaleY = 1) { const points = stroke.points; if (!points.length) return; ctx.save(); ctx.scale(scaleX, scaleY); ctx.globalAlpha = stroke.opacity; if (stroke.erasing) ctx.globalCompositeOperation = 'destination-out'; ctx.strokeStyle = stroke.color; ctx.fillStyle = stroke.color; ctx.lineWidth = stroke.size; ctx.lineCap = 'round'; ctx.lineJoin = 'round'; if (points.length === 1) { ctx.beginPath(); ctx.arc(points[0].x, points[0].y, stroke.size / 2, 0, Math.PI * 2); ctx.fill(); } else { ctx.beginPath(); ctx.moveTo(points[0].x, points[0].y); for (const p of points.slice(1)) ctx.lineTo(p.x,p.y); ctx.stroke(); } ctx.restore(); }
